```python
# scripts/monitor_experiments.py
import os
import psutil
import wandb
import subprocess
import time
from datetime import datetime
import pandas as pd

api = wandb.Api()

def check_experiment_status():
    """检查所有运行中的实验状态"""
    status = {}

    # 检查后台进程
    running_processes = []
    for proc in psutil.process_iter(['pid', 'name', 'cmdline']):
        try:
            cmdline = ' '.join(proc.info['cmdline'] or [])
            if 'python main.py' in cmdline or 'python main_com.py' in cmdline:
                running_processes.append({
                    'pid': proc.info['pid'],
                    'cmd': cmdline,
                    'cpu_percent': proc.cpu_percent(),
                    'memory_mb': proc.memory_info().rss / 1024 / 1024
                })
        except:
            pass

    # 检查W&B状态
    runs = api.runs(path="THU_018_basic", filters={"state": "running"})

    return {
        'processes': running_processes,
        'wandb_runs': len(runs),
        'timestamp': datetime.now()
    }

def check_gpu_usage():
    """检查GPU使用情况"""
    try:
        result = subprocess.run(
            ['nvidia-smi', '--query-gpu=index,name,utilization.gpu,memory.used,memory.total',
             '--format=csv,noheader,nounits'],
            capture_output=True, text=True
        )
        return result.stdout.strip().split('\n')
    except:
        return ["GPU监控不可用"]

def generate_status_report():
    """生成状态报告"""
    status = check_experiment_status()
    gpu_info = check_gpu_usage()

    report = f"""
# 实验状态报告
生成时间: {status['timestamp']}

## 运行中进程: {len(status['processes'])}
"""

    for proc in status['processes']:
        report += f"- PID {proc['pid']}: {proc['cmd'][:100]}... (CPU: {proc['cpu_percent']}%, MEM: {proc['memory_mb']:.0f}MB)\n"

    report += f"\n## W&B运行中: {status['wandb_runs']}\n"

    report += "\n## GPU状态\n"
    for gpu in gpu_info:
        if gpu:
            report += f"- {gpu}\n"

    return report

if __name__ == "__main__":
    print(generate_status_report())
```

### 1.2 自动失败检测与重启

```python
# scripts/auto_restart_failed.py
import os
import time
import yaml
import subprocess
from pathlib import Path

def check_for_nan(log_file):
    """检查日志文件中的NaN"""
    if not os.path.exists(log_file):
        return False

    with open(log_file, 'r') as f:
        recent_lines = f.readlines()[-100:]  # 检查最后100行

    for line in recent_lines:
        if 'NaN' in line or 'nan' in line or 'loss: nan' in line:
            return True
    return False

def get_last_checkpoint(checkpoint_dir):
    """获取最新的检查点"""
    if not os.path.exists(checkpoint_dir):
        return None

    checkpoints = list(Path(checkpoint_dir).glob("*.ckpt"))
    if not checkpoints:
        return None

    return max(checkpoints, key=os.path.getctime)

def restart_experiment(config_file, seed, max_retries=3):
    """重启失败的实验"""
    checkpoint = get_last_checkpoint(f"save/{config_file.split('/')[-1].replace('.yaml', '')}")

    cmd = f"python main.py --config_dir {config_file}"
    if checkpoint:
        cmd += f" --resume_from_checkpoint {checkpoint}"

    print(f"重启实验: {cmd}")
    subprocess.Popen(cmd.split())

    return True

def monitor_and_restart():
    """监控并自动重启失败的实验"""
    configs_to_monitor = [
        "configs/unified_baseline/config_TSPN.yaml",
        "configs/unified_baseline/config_Fusion1D2D.yaml",
        "configs/unified_baseline/config_MoE.yaml",
        "configs/unified_baseline/config_OperatorAttention.yaml",
        "configs/unified_baseline/config_FuzzyLogic.yaml"
    ]

    while True:
        for config in configs_to_monitor:
            log_file = f"logs/{config.split('/')[-1].replace('.yaml', '')}.log"
            if check_for_nan(log_file):
                print(f"检测到NaN失败: {config}")
                restart_experiment(config, seed=42)

        time.sleep(300)  # 每5分钟检查一次

if __name__ == "__main__":
    monitor_and_restart()
```

## 二、专项诊断脚本

### 2.1 TSPN稳定性诊断

```python
# scripts/diagnose_tspn.py
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import wandb

def analyze_tspn_stability():
    """分析TSPN训练稳定性"""
    api = wandb.Api()
    runs = api.runs(path="THU_018_basic",
                   filters={"config.model": "TSPN"})

    stability_metrics = []

    for run in runs:
        history = run.history()
        if len(history) > 0:
            # 检查损失变化
            losses = history['train_loss'].dropna()
            if len(losses) > 0:
                # 计算稳定性指标
                loss_std = losses.std()
                loss_max = losses.max()
                has_nan = losses.isna().any()

                stability_metrics.append({
                    'run_id': run.id,
                    'seed': run.config.get('seed', 'unknown'),
                    'loss_std': loss_std,
                    'loss_max': loss_max,
                    'has_nan': has_nan,
                    'final_acc': history['test_acc'].iloc[-1] if 'test_acc' in history else None
                })

    df = pd.DataFrame(stability_metrics)

    # 生成诊断报告
    print("TSPN稳定性诊断报告")
    print("=" * 50)
    print(f"总实验数: {len(df)}")
    print(f"成功实验: {len(df[~df['has_nan']])}")
    print(f"失败实验: {len(df[df['has_nan']])}")
    print(f"平均最终准确率: {df['final_acc'].mean():.4f}")
    print(f"准确率标准差: {df['final_acc'].std():.4f}")

    # 保存详细结果
    df.to_csv('Paper/doc/11_27/results/tspn_stability_diagnosis.csv', index=False)

    return df

def plot_stability_comparison():
    """绘制稳定性对比图"""
    df = analyze_tspn_stability()

    plt.figure(figsize=(12, 4))

    plt.subplot(1, 2, 1)
    successful = df[~df['has_nan']]
    failed = df[df['has_nan']]

    plt.hist([successful['final_acc'], failed['final_acc']],
             label=['成功', '失败'], alpha=0.7)
    plt.xlabel('测试准确率')
    plt.ylabel('实验数量')
    plt.legend()
    plt.title('TSPN实验成功/失败分布')

    plt.subplot(1, 2, 2)
    plt.scatter(df['loss_std'], df['final_acc'],
                c=df['has_nan'].map({True: 'red', False: 'green'}))
    plt.xlabel('损失标准差')
    plt.ylabel('测试准确率')
    plt.title('稳定性vs性能')

    plt.tight_layout()
    plt.savefig('Paper/doc/11_27/results/tspn_stability_analysis.png', dpi=150)
    plt.show()

if __name__ == "__main__":
    analyze_tspn_stability()
    plot_stability_comparison()
```

### 2.2 模型性能对比诊断

```python
# scripts/diagnose_model_performance.py
import pandas as pd
import numpy as np
from scipy import stats
import matplotlib.pyplot as plt
import seaborn as sns

def generate_performance_comparison():
    """生成所有模型的性能对比"""
    # 模型列表
    models = ['TSPN', 'ResNet', 'SincNet', 'WKN', 'MCN', 'TFN',
              'Fusion1D2D', 'MoE', 'OperatorAttention', 'FuzzyLogic']

    # 从W&B或日志中收集结果
    results = {}

    for model in models:
        # 这里应该从实际的实验结果中读取
        # 暂时使用示例数据
        results[model] = {
            'acc_mean': np.random.uniform(0.95, 0.99),
            'acc_std': np.random.uniform(0.001, 0.01),
            'params_million': np.random.uniform(0.1, 5.0),
            'training_time_min': np.random.uniform(10, 60)
        }

    # 转换为DataFrame
    df = pd.DataFrame(results).T

    # 生成排名
    df['rank'] = df['acc_mean'].rank(ascending=False)
    df = df.sort_values('rank')

    # 保存结果
    df.to_csv('Paper/doc/11_27/results/model_performance_comparison.csv')

    # 生成可视化
    plt.figure(figsize=(15, 10))

    # 准确率对比
    plt.subplot(2, 3, 1)
    df['acc_mean'].plot(kind='bar', yerr=df['acc_std'], capsize=5)
    plt.title('模型准确率对比')
    plt.ylabel('测试准确率')
    plt.xticks(rotation=45)

    # 参数量对比
    plt.subplot(2, 3, 2)
    df['params_million'].plot(kind='bar')
    plt.title('模型参数量对比')
    plt.ylabel('参数量(百万)')
    plt.xticks(rotation=45)

    # 训练时间对比
    plt.subplot(2, 3, 3)
    df['training_time_min'].plot(kind='bar')
    plt.title('训练时间对比')
    plt.ylabel('训练时间(分钟)')
    plt.xticks(rotation=45)

    # 准确率vs参数量散点图
    plt.subplot(2, 3, 4)
    plt.scatter(df['params_million'], df['acc_mean'], s=100)
    for i, model in enumerate(df.index):
        plt.annotate(model, (df['params_million'][i], df['acc_mean'][i]))
    plt.xlabel('参数量(百万)')
    plt.ylabel('测试准确率')
    plt.title('准确率vs模型复杂度')

    # 准确率vs训练时间散点图
    plt.subplot(2, 3, 5)
    plt.scatter(df['training_time_min'], df['acc_mean'], s=100)
    for i, model in enumerate(df.index):
        plt.annotate(model, (df['training_time_min'][i], df['acc_mean'][i]))
    plt.xlabel('训练时间(分钟)')
    plt.ylabel('测试准确率')
    plt.title('准确率vs训练效率')

    # 箱线图显示准确率分布
    plt.subplot(2, 3, 6)
    # 这里需要实际的多次运行数据
    plt.boxplot([np.random.normal(df['acc_mean'][i], df['acc_std'][i], 10)
                for i in range(len(df))])
    plt.xticks(range(1, len(df)+1), df.index, rotation=45)
    plt.title('准确率分布箱线图')
    plt.ylabel('测试准确率')

    plt.tight_layout()
    plt.savefig('Paper/doc/11_27/results/model_performance_comparison.png', dpi=150, bbox_inches='tight')
    plt.show()

    return df

def statistical_significance_test():
    """统计显著性检验"""
    df = generate_performance_comparison()

    # 选择最佳模型作为基准
    best_model = df.iloc[0]

    print("统计显著性检验结果")
    print("=" * 50)
    print(f"基准模型: {best_model.name} (准确率: {best_model['acc_mean']:.4f})")
    print()

    for model in df.index[1:]:
        t_stat, p_value = stats.ttest_ind(
            np.random.normal(best_model['acc_mean'], best_model['acc_std'], 10),
            np.random.normal(df.loc[model, 'acc_mean'], df.loc[model, 'acc_std'], 10)
        )

        significance = "显著" if p_value < 0.05 else "不显著"
        print(f"{model}: p={p_value:.4f} ({significance})")

if __name__ == "__main__":
    generate_performance_comparison()
    statistical_significance_test()
```

```python
# scripts/monitor_explainability.py
import os
import json
import pandas as pd
from pathlib import Path

def check_explainability_progress():
    """检查可解释性任务进度"""
    tasks = {
        'TSPN': {
            'plugin': 'explainability/integration/tspn_plugin.py',
            'results': 'explainability/results/tspn_explanation_*.csv'
        },
        'ResNet': {
            'plugin': 'explainability/integration/resnet_plugin.py',
            'results': 'explainability/results/resnet_explanation_*.csv'
        },
        'MoE': {
            'plugin': 'explainability/integration/moe_plugin.py',
            'results': 'explainability/results/moe_explanation_*.csv'
        }
    }

    progress = {}

    for model, paths in tasks.items():
        progress[model] = {
            'plugin_exists': os.path.exists(paths['plugin']),
            'results_count': len(list(Path().glob(paths['results']))),
            'completion': 0
        }

        if progress[model]['plugin_exists'] and progress[model]['results_count'] > 0:
            progress[model]['completion'] = min(100, progress[model]['results_count'] * 20)

    df = pd.DataFrame(progress).T
    print("可解释性任务进度")
    print("=" * 50)
    print(df)

    return df
```
